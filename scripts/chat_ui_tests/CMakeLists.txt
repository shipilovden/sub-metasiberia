cmake_minimum_required(VERSION 3.16)
project(ChatOverlayTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
add_executable(chat_catalog_test catalog.cpp)
add_executable(chat_layout_test layout.cpp)
add_executable(chat_read_state_test read_state.cpp)
if(MSVC)
  target_compile_options(chat_catalog_test PRIVATE /utf-8 /W4)
  target_compile_options(chat_layout_test PRIVATE /utf-8 /W4)
  target_compile_options(chat_read_state_test PRIVATE /utf-8 /W4)
endif()
add_test(NAME catalog COMMAND chat_catalog_test)
add_test(NAME layout COMMAND chat_layout_test)
add_test(NAME read_state COMMAND chat_read_state_test)
